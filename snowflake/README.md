```sql
USE ROLE ACCOUNTADMIN;

CREATE DATABASE IF NOT EXISTS DEMO_DB;
CREATE SCHEMA IF NOT EXISTS DEMO_DB.SALES;

CREATE OR REPLACE TABLE DEMO_DB.SALES.CUSTOMER (
    CUSTOMER_ID NUMBER,
    CUSTOMER_NAME STRING,
    EMAIL STRING
);

INSERT INTO DEMO_DB.SALES.CUSTOMER VALUES
    (1, 'Ravi',  'ravi@example.com'),
    (2, 'Priya', 'priya@example.com'),
    (3, 'John',  'john@example.com');


USE ROLE USERADMIN;

CREATE ROLE MASKING_ADMIN;
CREATE ROLE MASKING_ANALYST;

SELECT CURRENT_USER();

GRANT ROLE MASKING_ADMIN
TO USER MAILTOGOPS;

GRANT ROLE MASKING_ANALYST
TO USER MAILTOGOPS;

USE ROLE SECURITYADMIN;

GRANT USAGE ON DATABASE DEMO_DB
TO ROLE MASKING_ADMIN;

GRANT USAGE ON SCHEMA DEMO_DB.SALES
TO ROLE MASKING_ADMIN;

GRANT SELECT ON TABLE DEMO_DB.SALES.CUSTOMER
TO ROLE MASKING_ADMIN;


GRANT USAGE ON DATABASE DEMO_DB
TO ROLE MASKING_ANALYST;

GRANT USAGE ON SCHEMA DEMO_DB.SALES
TO ROLE MASKING_ANALYST;

GRANT SELECT ON TABLE DEMO_DB.SALES.CUSTOMER
TO ROLE MASKING_ANALYST;

USE ROLE ACCOUNTADMIN;

CREATE OR REPLACE MASKING POLICY
DEMO_DB.SALES.EMAIL_MASK
AS (VAL STRING)
RETURNS STRING ->
    CASE
        WHEN CURRENT_ROLE() = 'MASKING_ADMIN'
            THEN VAL
        ELSE '********'
    END;


    ALTER TABLE DEMO_DB.SALES.CUSTOMER
MODIFY COLUMN EMAIL
SET MASKING POLICY DEMO_DB.SALES.EMAIL_MASK;

-----
USE ROLE MASKING_ANALYST;

SELECT CURRENT_USER(), CURRENT_ROLE();

SELECT *
FROM DEMO_DB.SALES.CUSTOMER;

----
USE ROLE MASKING_ADMIN;

SELECT CURRENT_USER(), CURRENT_ROLE();

SELECT *
FROM DEMO_DB.SALES.CUSTOMER;

```
