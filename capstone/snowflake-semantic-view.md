```
CREATE OR REPLACE DATABASE AI_DEMO;

CREATE OR REPLACE SCHEMA AI_DEMO.SALES;

USE DATABASE AI_DEMO;
USE SCHEMA SALES;


CREATE OR REPLACE TABLE CUSTOMERS (
    CUSTOMER_ID   NUMBER,
    CUSTOMER_NAME VARCHAR,
    REGION        VARCHAR
);


CREATE OR REPLACE TABLE PRODUCTS (
    PRODUCT_ID    NUMBER,
    PRODUCT_NAME  VARCHAR,
    CATEGORY      VARCHAR
);


CREATE OR REPLACE TABLE SALES (
    SALE_ID       NUMBER,
    CUSTOMER_ID   NUMBER,
    PRODUCT_ID    NUMBER,
    SALE_DATE     DATE,
    QUANTITY      NUMBER,
    UNIT_PRICE    NUMBER(10,2)
);
```
